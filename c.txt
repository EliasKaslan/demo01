"""Batch invariance test with environment variable control."""

import os
import random
import time

import pytest
import torch
from vllm import LLM, SamplingParams

DEFAULT_MODEL = "/home/data/Qwen3-1.7B"

os.environ['OMP_NUM_THREADS'] = '1'
os.environ['MKL_NUM_THREADS'] = '1'


def _random_prompt(min_words: int = 10, max_words: int = 50) -> str:
    """生成固定种子的伪随机 prompt（短文本，确保生成 token）"""
    templates = [
        "Question: What is the capital of France?\nAnswer: The capital of France is",
        "Q: How does photosynthesis work?\nA: Photosynthesis is the process by which",
        "User: Can you explain quantum mechanics?\nAssistant: Quantum mechanics is",
        "Once upon a time in a distant galaxy, there lived",
        "To implement a binary search tree in Python, first we need to",
        "Climate change is caused by several factors including",
    ]
    base = random.choice(templates)
    target = random.randint(min_words, max_words)
    if target > 50:
        base += " This is an interesting topic that deserves more explanation. " * (target // 50)
    return base


def _extract_step_logprobs(request_output):
    """从 RequestOutput 提取每一步的 logprob 张量及 token id 列表"""
    if getattr(request_output, "outputs", None):
        inner = request_output.outputs[0]
        if hasattr(inner, "logprobs") and inner.logprobs is not None:
            t = torch.tensor(
                [inner.logprobs[i][tid].logprob for i, tid in enumerate(inner.token_ids)],
                dtype=torch.float32,
            )
            return t, inner.token_ids
    return None, None


def test_logprobs_bitwise_batch_invariance_bs1_vs_bsN(monkeypatch: pytest.MonkeyPatch):
    """
    验证 batch 不变性：BS=1 与 BS=N 时，相同 prompt 应输出相同的 logprobs。
    参数全部通过环境变量控制（支持整数/浮点数）：
        VLLM_TEST_TP_SIZE          - TP 尺寸 (默认 1)
        VLLM_TEST_TEMPERATURE      - 采样温度 (默认 0.0)
        VLLM_TEST_MAX_TOKENS       - 最大生成 token 数 (默认 50)
        VLLM_TEST_MAX_PROMPT_NUM   - 每批最大 prompt 数量 (默认 64)
        VLLM_TEST_SEED             - 随机种子 (默认 12345)
        VLLM_TEST_MODEL            - 模型路径 (默认 /home/data/Qwen3-1.7B)
    """
    # ---------- 从环境变量读取参数 ----------
    tp_size = int(os.getenv("VLLM_TEST_TP_SIZE", "1"))
    temperature = float(os.getenv("VLLM_TEST_TEMPERATURE", "0.0"))
    max_tokens = int(os.getenv("VLLM_TEST_MAX_TOKENS", "50"))
    max_prompt_num = int(os.getenv("VLLM_TEST_MAX_PROMPT_NUM", "64"))
    max_prompt_len = int(os.getenv("VLLM_TEST_MAX_PROMPT_LEN", "0"))
    seed = int(os.getenv("VLLM_TEST_SEED", "12345"))
    model_name = os.getenv("VLLM_TEST_MODEL", DEFAULT_MODEL)
    max_num_batched_tokens = int(os.getenv("VLLM_TEST_MAX_NUM_BATCHED_TOKENS", "65536"))
    enable_prefix_caching = os.getenv("VLLM_TEST_ENABLE_PREFIX_CACHING", "False").lower() in ("true", "1", "yes")
    

    random.seed(seed)

    # ---------- 启用 batch_invariant 模式（禁用 custom all-reduce）----------
    # try:
    #     from vllm.model_executor.layers.batch_invariant import vllm_is_batch_invariant
    #     disable_custom_ar = vllm_is_batch_invariant()
    # except ImportError:
    #     disable_custom_ar = True

    # if Fasle:
    #     print(f"\n{'=' * 80}")
    #     print(f"BATCH INVARIANCE MODE: Disabling custom all-reduce (TP={tp_size})")
    #     print(f"{'=' * 80}\n")

    # ---------- 初始化 LLM ----------
    try:
        llm = LLM(
            model=model_name,
            tensor_parallel_size=tp_size,
            enable_prefix_caching=enable_prefix_caching,
            max_num_seqs=max_prompt_num,
            max_model_len=22528,
            dtype="bfloat16",
            gpu_memory_utilization=0.9,
            enforce_eager=True,
            distributed_executor_backend="mp",
            max_num_batched_tokens=max_num_batched_tokens,
            load_format="dummy",
        )
    except Exception as e:
        pytest.fail(f"LLM initialization failed: {str(e)}")

    # ---------- 生成 prompts ----------
    prompts = [_random_prompt(max_prompt_len + 10, max_prompt_len + 50) for _ in range(max_prompt_num)]

    # ---------- 采样参数 ----------
    sp = SamplingParams(
        temperature=temperature,
        max_tokens=max_tokens,
        seed=seed,
        logprobs=5,
    )

    # ---------- BS=1 逐个运行 ----------
    print("\n" + "=" * 80)
    print("STARTING BS=1 RUNS (each prompt individually)")
    print("=" * 80 + "\n")

    # experimental_config = torch.profiler._ExperimentalConfig()
    # # with torch.profiler.profile(
    # #     activities=[
    # #             torch.profiler.ProfilerActivity.CPU,
    # #             torch.profiler.ProfilerActivity.CUDA
    # #         ],
    # #     schedule=torch.profiler.schedule(wait=0, warmup=0, active=1, repeat=1, skip_first=0),
    # #     experimental_config = experimental_config,
    # #     on_trace_ready=torch.profiler.tensorboard_trace_handler("/home/w00899129/vllm_batch_inv_v2/dsv3_profiling_0_49")
    # # ) as prof:
    # import torch_npu
    # with torch_npu.profiler.profile(
    # activities=[
    #     torch_npu.profiler.ProfilerActivity.CPU,
    #     torch_npu.profiler.ProfilerActivity.NPU
    #     ],
    # schedule=torch_npu.profiler.schedule(wait=0, warmup=0, active=1, repeat=1, skip_first=0),
    # on_trace_ready=torch_npu.profiler.tensorboard_trace_handler("/data/w00899129/vllm_profile_eager_qwen3-1.7b-npu_0.14.0"),
    # record_shapes=True,
    # profile_memory=False,
    # # 关键修改：启用调用栈跟踪
    # with_stack=True,      # 改为 True
    # with_modules=True,    # 改为 True，获取模块信息
    # with_flops=False,
    # experimental_config=experimental_config) as prof:
    bs1_logprobs = []
    bs1_tokens = []
    for idx, p in enumerate(prompts):
        print(f"\n[BS=1] Running prompt {idx}/{len(prompts)} - Preview: {p[:80]}")
        try:
            outs = llm.generate([p], sp, use_tqdm=False)
            assert len(outs) == 1
            step_logprobs, token_ids = _extract_step_logprobs(outs[0])
            if step_logprobs is None:
                pytest.skip("Logits are not available on RequestOutput; "
                            "enable logprobs return to run this test.")
            bs1_logprobs.append(step_logprobs)
            bs1_tokens.append(token_ids)
            print(f"[BS=1] Prompt {idx} generated tokens: {token_ids}")
        except Exception as e:
            pytest.fail(f"BS=1 generation failed for prompt {idx}: {str(e)}")

    # ---------- BS=N 批量运行 ----------
    print("\n" + "=" * 80)
    print(f"STARTING BS={len(prompts)} RUN (all prompts batched)")
    print("=" * 80 + "\n")

    try:
        outs_batch = llm.generate(prompts, sp, use_tqdm=False)
        assert len(outs_batch) == len(prompts)
    except Exception as e:
        pytest.fail(f"BS=N generation failed: {str(e)}")
    # prof.step()
    
    bsN_logprobs = []
    bsN_tokens = []

    print(f"\n[BS={len(prompts)}] Processing batched outputs...")
    for idx, o in enumerate(outs_batch):
        tokens = o.outputs[0].token_ids if o.outputs else "N/A"
        print(f"[BS={len(prompts)}] Prompt {idx} generated tokens: {tokens}")
        step_logprobs, token_ids = _extract_step_logprobs(o)
        if step_logprobs is None:
            pytest.skip("Logits are not available on RequestOutput; "
                        "enable logprobs return to run this test.")
        bsN_logprobs.append(step_logprobs)
        bsN_tokens.append(token_ids)

    # ---------- 释放资源 ----------
    del llm
    # 增加等待时间，确保GPU资源完全释放
    time.sleep(5)
    # cleanup_dist_env_and_memory()

    # ---------- 详细失败信息收集 ----------
    failed_prompts = []
    failure_stats = {
        "total": 0,
        "step_mismatch": 0,
        "token_mismatch": 0,
        "logprob_mismatch": 0,
        "other": 0
    }

    for i, (lp1, lpN, tk1, tkN) in enumerate(
        zip(bs1_logprobs, bsN_logprobs, bs1_tokens, bsN_tokens)
    ):
        # 1. 检查步数是否一致
        if len(lp1) != len(lpN):
            failure_stats["step_mismatch"] += 1
            failure_stats["total"] += 1
            failed_prompts.append({
                "prompt_idx": i,
                "failure_type": "step_mismatch",
                "reason": f"Different number of steps: {len(lp1)} (BS=1) vs {len(lpN)} (BS=N)",
                "prompt_preview": prompts[i][:100],
                "bs1_tokens": tk1,
                "bsN_tokens": tkN,
            })
            continue

        # 2. 检查采样 token 是否一致
        if tk1 != tkN:
            failure_stats["token_mismatch"] += 1
            failure_stats["total"] += 1
            
            # 找出第一个不同的位置
            diff_pos = -1
            for pos, (t1, tN) in enumerate(zip(tk1, tkN)):
                if t1 != tN:
                    diff_pos = pos
                    break
            
            failed_prompts.append({
                "prompt_idx": i,
                "failure_type": "token_mismatch",
                "reason": f"Different tokens sampled at position {diff_pos}",
                "prompt_preview": prompts[i][:100],
                "bs1_tokens": tk1,
                "bsN_tokens": tkN,
                "diff_position": diff_pos,
                "bs1_token_at_diff": tk1[diff_pos] if diff_pos != -1 and diff_pos < len(tk1) else None,
                "bsN_token_at_diff": tkN[diff_pos] if diff_pos != -1 and diff_pos < len(tkN) else None,
            })
            continue

        # 3. 逐 token 比较 logprob（仅当 tokens 完全一致时）
        logprob_mismatch_found = False
        max_diff = 0.0
        mismatch_pos = -1
        
        for t, (a, b) in enumerate(zip(lp1, lpN)):
            if not torch.equal(a, b):
                diff = torch.abs(a - b).max().item()
                if diff > max_diff:
                    max_diff = diff
                    mismatch_pos = t
                logprob_mismatch_found = True
        
        if logprob_mismatch_found:
            failure_stats["logprob_mismatch"] += 1
            failure_stats["total"] += 1
            failed_prompts.append({
                "prompt_idx": i,
                "failure_type": "logprob_mismatch",
                "reason": f"Logprob bitwise mismatch (max_diff={max_diff:.6e} at pos {mismatch_pos})",
                "prompt_preview": prompts[i][:100],
                "bs1_tokens": tk1,
                "bsN_tokens": tkN,
                "max_diff": max_diff,
                "mismatch_position": mismatch_pos,
            })

    # ---------- 失败汇总与断言 ----------
    if failed_prompts:
        print(f"\n{'=' * 80}")
        fail_msg = f"BATCH INVARIANCE FAILURES: {failure_stats['total']}/{len(prompts)} prompts failed"
        print(fail_msg)
        print(f"{'=' * 80}")
        
        # 打印统计汇总
        print("\nFAILURE STATISTICS:")
        print(f"  - Step Mismatch: {failure_stats['step_mismatch']}")
        print(f"  - Token Mismatch: {failure_stats['token_mismatch']}")
        print(f"  - Logprob Mismatch: {failure_stats['logprob_mismatch']}")
        print(f"  - Other: {failure_stats['other']}")
        
        # 打印详细的失败信息（最多显示前5个）
        print("\nDETAILED FAILURES (first 5):")
        for fail in failed_prompts[:5]:
            print(f"\nPrompt {fail['prompt_idx']}:")
            print(f"  Type: {fail['failure_type']}")
            print(f"  Reason: {fail['reason']}")
            print(f"  Preview: {fail['prompt_preview'][:80]}...")
            if fail['failure_type'] == 'token_mismatch':
                print(f"  BS=1 token at diff: {fail.get('bs1_token_at_diff')}")
                print(f"  BS=N token at diff: {fail.get('bsN_token_at_diff')}")
            elif fail['failure_type'] == 'logprob_mismatch':
                print(f"  Max diff: {fail.get('max_diff', 0):.6e}")
        
        print(f"{'=' * 80}\n")
        
        # 将失败统计信息保存到环境变量，供shell脚本读取
        with open("batch_invariant_failure_stats.txt", "w") as f:
            f.write(f"TOTAL_FAILURES={failure_stats['total']}\n")
            f.write(f"STEP_MISMATCH={failure_stats['step_mismatch']}\n")
            f.write(f"TOKEN_MISMATCH={failure_stats['token_mismatch']}\n")
            f.write(f"LOGPROB_MISMATCH={failure_stats['logprob_mismatch']}\n")
            f.write(f"FIRST_FAILURE_REASON={failed_prompts[0]['reason'] if failed_prompts else ''}\n")
        
        # pytest.fail(fail_msg)
    else:
        # 没有失败，删除统计文件（如果存在）
        if os.path.exists("batch_invariant_failure_stats.txt"):
            os.remove("batch_invariant_failure_stats.txt")

apt update && apt install -y wget tar bzip2 gcc g++ make libssl-dev libffi-dev

cd /mnt/share/x00886548/tmp
wget https://mirrors.tuna.tsinghua.edu.cn/anaconda/miniconda/Miniconda3-latest-Linux-aarch64.sh

bash Miniconda3-latest-Linux-aarch64.sh -b -p /mnt/share/x00886548/miniconda3

# 永久生效，重启容器不用重新source
echo "source /mnt/share/x00886548/miniconda3/etc/profile.d/conda.sh" >> /etc/profile
source /etc/profile

# 写入conda配置，指定envs、pkgs目录到挂载盘
cat > ~/.condarc <<EOF
envs_dirs:
  - /mnt/share/vllm/conda_envs
pkgs_dirs:
  - /mnt/share/vllm/conda_pkgs
show_channel_urls: true
default_channels:
  - https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/main
  - https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/r
channel_alias: https://mirrors.tuna.tsinghua.edu.cn/anaconda
EOF

source /mnt/share/x00886548/miniconda3/etc/profile.d/conda.sh

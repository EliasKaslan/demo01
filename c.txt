"""BS=1 与 BS=N 的 batch 不变性测试：相同 prompt 的逐 token logprobs 应位级一致。

所有测试参数均通过环境变量注入（便于外层 test_batch_inv.sh 做全组合遍历）：

    VLLM_TEST_MODEL                   模型路径（默认 /home/data/Qwen3-1.7B）
    VLLM_TEST_TP_SIZE                 TP 并行度（默认 1）
    VLLM_TEST_TEMPERATURE             采样温度（默认 0.0）
    VLLM_TEST_MAX_TOKENS              每请求最大生成 token 数（默认 50）
    VLLM_TEST_MAX_PROMPT_NUM          prompt 数量，即 BS=N 的 N（默认 64）
    VLLM_TEST_MAX_PROMPT_LEN          prompt 词数基准，实际词数在其 +10 ~ +50（默认 0）
    VLLM_TEST_MAX_NUM_BATCHED_TOKENS  max_num_batched_tokens（默认 65536）
    VLLM_TEST_ENABLE_PREFIX_CACHING   是否开启 prefix cache（默认 False）
    VLLM_TEST_SEED                    随机种子（默认 12345）

与外层 shell 脚本的约定（修改前请同步 test_batch_inv.sh）：
    1. 失败时向 stdout 打印含 "BATCH INVARIANCE FAILURES: X/N prompts failed"
       的汇总行，并输出 "  Reason: ..." 形式的失败原因；
    2. 失败时将统计信息写入当前目录的 batch_invariant_failure_stats.txt；
    3. 成功时确保该统计文件不存在；
    4. pytest.fail 有意保持禁用：失败用例仍以退出码 0 结束，由 shell 侧
       通过日志关键字判定 FAIL 并读取统计文件。
"""

import os
import random
import time

import pytest
import torch
from vllm import LLM, SamplingParams

DEFAULT_MODEL = "/home/data/Qwen3-1.7B"

# 失败统计文件路径（由外层 test_batch_inv.sh 解析，勿改名）
STATS_FILE = "batch_invariant_failure_stats.txt"

# 限制 CPU 线程数，避免主机侧线程调度抖动引入非确定性
os.environ["OMP_NUM_THREADS"] = "1"
os.environ["MKL_NUM_THREADS"] = "1"


# ---------- 工具函数 ----------


def _load_test_config() -> dict:
    """从环境变量读取全部测试参数。"""
    return {
        "model": os.getenv("VLLM_TEST_MODEL", DEFAULT_MODEL),
        "tp_size": int(os.getenv("VLLM_TEST_TP_SIZE", "1")),
        "temperature": float(os.getenv("VLLM_TEST_TEMPERATURE", "0.0")),
        "max_tokens": int(os.getenv("VLLM_TEST_MAX_TOKENS", "50")),
        "max_prompt_num": int(os.getenv("VLLM_TEST_MAX_PROMPT_NUM", "64")),
        "max_prompt_len": int(os.getenv("VLLM_TEST_MAX_PROMPT_LEN", "0")),
        "max_num_batched_tokens": int(os.getenv("VLLM_TEST_MAX_NUM_BATCHED_TOKENS", "65536")),
        "enable_prefix_caching": os.getenv("VLLM_TEST_ENABLE_PREFIX_CACHING", "False").lower()
        in ("true", "1", "yes"),
        "seed": int(os.getenv("VLLM_TEST_SEED", "12345")),
    }


def _random_prompt(min_words: int = 10, max_words: int = 50) -> str:
    """生成固定种子的伪随机 prompt（模板 + 重复填充文本）。"""
    templates = [
        "Question: What is the capital of France?\nAnswer: The capital of France is",
        "Q: How does photosynthesis work?\nA: Photosynthesis is the process by which",
        "User: Can you explain quantum mechanics?\nAssistant: Quantum mechanics is",
        "Once upon a time in a distant galaxy, there lived",
        "To implement a binary search tree in Python, first we need to",
        "Climate change is caused by several factors including",
    ]
    base = random.choice(templates)
    target = random.randint(min_words, max_words)
    if target > 50:
        base += " This is an interesting topic that deserves more explanation. " * (target // 50)
    return base


def _extract_step_logprobs(request_output):
    """从 RequestOutput 提取每步采样 token 的 logprob 张量及 token id 列表。"""
    if getattr(request_output, "outputs", None):
        inner = request_output.outputs[0]
        if hasattr(inner, "logprobs") and inner.logprobs is not None:
            step_logprobs = torch.tensor(
                [inner.logprobs[i][tid].logprob for i, tid in enumerate(inner.token_ids)],
                dtype=torch.float32,
            )
            return step_logprobs, inner.token_ids
    return None, None


def _first_token_diff(tokens_a, tokens_b) -> int:
    """返回第一处不一致的 token 位置；完全一致返回 -1。"""
    for pos, (a, b) in enumerate(zip(tokens_a, tokens_b)):
        if a != b:
            return pos
    return -1


def _scan_logprob_diffs(lp1, lpN):
    """逐 step 位级比较 logprob，返回 (是否存在不一致, 最大差值, 最大差值所在 step)。"""
    has_mismatch = False
    max_diff = 0.0
    mismatch_pos = -1
    for t, (a, b) in enumerate(zip(lp1, lpN)):
        if not torch.equal(a, b):
            has_mismatch = True
            diff = torch.abs(a - b).max().item()
            if diff > max_diff:
                max_diff = diff
                mismatch_pos = t
    return has_mismatch, max_diff, mismatch_pos


# ---------- 结果对比与报告 ----------


def _compare_runs(prompts, bs1_logprobs, bsN_logprobs, bs1_tokens, bsN_tokens):
    """逐 prompt 对比 BS=1 与 BS=N 的输出，返回 (失败详情列表, 统计字典)。"""
    failed_prompts = []
    failure_stats = {
        "total": 0,
        "step_mismatch": 0,
        "token_mismatch": 0,
        "logprob_mismatch": 0,
        "other": 0,
    }

    for i, (lp1, lpN, tk1, tkN) in enumerate(zip(bs1_logprobs, bsN_logprobs, bs1_tokens, bsN_tokens)):
        # 1. 步数一致性
        if len(lp1) != len(lpN):
            failure_stats["step_mismatch"] += 1
            failure_stats["total"] += 1
            failed_prompts.append(
                {
                    "prompt_idx": i,
                    "failure_type": "step_mismatch",
                    "reason": f"Different number of steps: {len(lp1)} (BS=1) vs {len(lpN)} (BS=N)",
                    "prompt_preview": prompts[i][:100],
                    "bs1_tokens": tk1,
                    "bsN_tokens": tkN,
                }
            )
            continue

        # 2. 采样 token 一致性
        if tk1 != tkN:
            diff_pos = _first_token_diff(tk1, tkN)
            failure_stats["token_mismatch"] += 1
            failure_stats["total"] += 1
            failed_prompts.append(
                {
                    "prompt_idx": i,
                    "failure_type": "token_mismatch",
                    "reason": f"Different tokens sampled at position {diff_pos}",
                    "prompt_preview": prompts[i][:100],
                    "bs1_tokens": tk1,
                    "bsN_tokens": tkN,
                    "diff_position": diff_pos,
                    "bs1_token_at_diff": tk1[diff_pos] if 0 <= diff_pos < len(tk1) else None,
                    "bsN_token_at_diff": tkN[diff_pos] if 0 <= diff_pos < len(tkN) else None,
                }
            )
            continue

        # 3. 逐 token logprob 位级一致性（仅当 tokens 完全一致时才比较）
        has_mismatch, max_diff, mismatch_pos = _scan_logprob_diffs(lp1, lpN)
        if has_mismatch:
            failure_stats["logprob_mismatch"] += 1
            failure_stats["total"] += 1
            failed_prompts.append(
                {
                    "prompt_idx": i,
                    "failure_type": "logprob_mismatch",
                    "reason": f"Logprob bitwise mismatch (max_diff={max_diff:.6e} at pos {mismatch_pos})",
                    "prompt_preview": prompts[i][:100],
                    "bs1_tokens": tk1,
                    "bsN_tokens": tkN,
                    "max_diff": max_diff,
                    "mismatch_position": mismatch_pos,
                }
            )

    return failed_prompts, failure_stats


def _report_failures(prompts, failed_prompts, failure_stats):
    """打印失败报告，并将统计信息写入 STATS_FILE 供外层 shell 脚本解析。"""
    if not failed_prompts:
        # 无失败：确保统计文件不存在
        if os.path.exists(STATS_FILE):
            os.remove(STATS_FILE)
        return

    print(f"\n{'=' * 80}")
    fail_msg = f"BATCH INVARIANCE FAILURES: {failure_stats['total']}/{len(prompts)} prompts failed"
    print(fail_msg)
    print(f"{'=' * 80}")

    # 统计汇总
    print("\nFAILURE STATISTICS:")
    print(f"  - Step Mismatch: {failure_stats['step_mismatch']}")
    print(f"  - Token Mismatch: {failure_stats['token_mismatch']}")
    print(f"  - Logprob Mismatch: {failure_stats['logprob_mismatch']}")
    print(f"  - Other: {failure_stats['other']}")

    # 详细失败信息（最多显示前 5 个）
    print("\nDETAILED FAILURES (first 5):")
    for fail in failed_prompts[:5]:
        print(f"\nPrompt {fail['prompt_idx']}:")
        print(f"  Type: {fail['failure_type']}")
        print(f"  Reason: {fail['reason']}")
        print(f"  Preview: {fail['prompt_preview'][:80]}...")
        if fail["failure_type"] == "token_mismatch":
            print(f"  BS=1 token at diff: {fail.get('bs1_token_at_diff')}")
            print(f"  BS=N token at diff: {fail.get('bsN_token_at_diff')}")
        elif fail["failure_type"] == "logprob_mismatch":
            print(f"  Max diff: {fail.get('max_diff', 0):.6e}")

    print(f"{'=' * 80}\n")

    # 将失败统计写入文件，供外层 shell 脚本（test_batch_inv.sh）读取
    with open(STATS_FILE, "w") as f:
        f.write(f"TOTAL_FAILURES={failure_stats['total']}\n")
        f.write(f"STEP_MISMATCH={failure_stats['step_mismatch']}\n")
        f.write(f"TOKEN_MISMATCH={failure_stats['token_mismatch']}\n")
        f.write(f"LOGPROB_MISMATCH={failure_stats['logprob_mismatch']}\n")
        f.write(f"FIRST_FAILURE_REASON={failed_prompts[0]['reason']}\n")

    # 注意：此处有意不调用 pytest.fail。失败用例仍以退出码 0 结束，由外层
    # shell 脚本通过日志关键字 "BATCH INVARIANCE FAILURES:" 判定 FAIL，并从
    # STATS_FILE 读取细粒度统计。如需恢复 pytest 断言行为，取消下行注释：
    # pytest.fail(fail_msg)


# ---------- 测试入口 ----------


def test_logprobs_bitwise_batch_invariance_bs1_vs_bsN(monkeypatch: pytest.MonkeyPatch):
    """验证 batch 不变性：BS=1 与 BS=N 下，相同 prompt 应产生位级一致的 logprobs。"""
    cfg = _load_test_config()
    random.seed(cfg["seed"])

    # 打印生效配置，便于从日志中复现单次运行
    print("\n" + "=" * 80)
    print("TEST CONFIG:")
    for key, value in cfg.items():
        print(f"  {key}: {value}")
    print("=" * 80)

    # ---------- 初始化 LLM ----------
    try:
        llm = LLM(
            model=cfg["model"],
            tensor_parallel_size=cfg["tp_size"],
            enable_prefix_caching=cfg["enable_prefix_caching"],
            max_num_seqs=cfg["max_prompt_num"],
            max_model_len=22528,
            dtype="bfloat16",
            gpu_memory_utilization=0.9,
            enforce_eager=True,
            distributed_executor_backend="mp",
            max_num_batched_tokens=cfg["max_num_batched_tokens"],
            load_format="dummy",
        )
    except Exception as e:
        pytest.fail(f"LLM initialization failed: {str(e)}")

    # ---------- 生成 prompts 与采样参数 ----------
    min_words = cfg["max_prompt_len"] + 10
    max_words = cfg["max_prompt_len"] + 50
    prompts = [_random_prompt(min_words, max_words) for _ in range(cfg["max_prompt_num"])]

    sp = SamplingParams(
        temperature=cfg["temperature"],
        max_tokens=cfg["max_tokens"],
        seed=cfg["seed"],
        logprobs=5,
    )

    # ---------- BS=1：逐条运行 ----------
    print("\n" + "=" * 80)
    print("STARTING BS=1 RUNS (each prompt individually)")
    print("=" * 80 + "\n")

    bs1_logprobs = []
    bs1_tokens = []
    for idx, prompt in enumerate(prompts):
        print(f"\n[BS=1] Running prompt {idx}/{len(prompts)} - Preview: {prompt[:80]}")
        try:
            outs = llm.generate([prompt], sp, use_tqdm=False)
            assert len(outs) == 1
            step_logprobs, token_ids = _extract_step_logprobs(outs[0])
            if step_logprobs is None:
                pytest.skip(
                    "Logits are not available on RequestOutput; "
                    "enable logprobs return to run this test."
                )
            bs1_logprobs.append(step_logprobs)
            bs1_tokens.append(token_ids)
            print(f"[BS=1] Prompt {idx} generated tokens: {token_ids}")
        except Exception as e:
            pytest.fail(f"BS=1 generation failed for prompt {idx}: {str(e)}")

    # ---------- BS=N：批量运行 ----------
    print("\n" + "=" * 80)
    print(f"STARTING BS={len(prompts)} RUN (all prompts batched)")
    print("=" * 80 + "\n")

    try:
        outs_batch = llm.generate(prompts, sp, use_tqdm=False)
        assert len(outs_batch) == len(prompts)
    except Exception as e:
        pytest.fail(f"BS=N generation failed: {str(e)}")

    bsN_logprobs = []
    bsN_tokens = []
    print(f"\n[BS={len(prompts)}] Processing batched outputs...")
    for idx, output in enumerate(outs_batch):
        tokens = output.outputs[0].token_ids if output.outputs else "N/A"
        print(f"[BS={len(prompts)}] Prompt {idx} generated tokens: {tokens}")
        step_logprobs, token_ids = _extract_step_logprobs(output)
        if step_logprobs is None:
            pytest.skip(
                "Logits are not available on RequestOutput; "
                "enable logprobs return to run this test."
            )
        bsN_logprobs.append(step_logprobs)
        bsN_tokens.append(token_ids)

    # ---------- 释放资源 ----------
    del llm
    time.sleep(5)  # 等待 NPU/显存资源完全释放

    # ---------- 对比与报告 ----------
    failed_prompts, failure_stats = _compare_runs(
        prompts, bs1_logprobs, bsN_logprobs, bs1_tokens, bsN_tokens
    )
    _report_failures(prompts, failed_prompts, failure_stats)
